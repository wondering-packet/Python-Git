SELECT
->     d.hostname,
->     INET_NTOA(d.ip_addr) AS ip_address,
->     d.description,
->     l.name AS location_name,
->     r.name AS rack_name,
->     r.size AS rack_size,
->     d.rack_start AS rack_start_position
-> FROM
->     devices d
-> LEFT JOIN
->     locations l ON d.location = l.id
-> LEFT JOIN
->     racks r ON d.rack = r.id
-> ORDER BY
->     d.hostname ASC;
